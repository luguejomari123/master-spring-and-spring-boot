--name mysql    = container name
--publish 3306:3306  = port to publish

image = package representing specific version of software
container = rruntime instance of a docker image
-p 5000:5000   used to map/expose port on different port    port on machine(host):port of original image(internal/only can be changed by dev)        hostport:containerport
-d = detached mode, so you can still use terminal after running a container
dockerfile = file with instruction to create a docker image

2. building spring boot docker image using multi stage dockerfile (does not use anything in local machine to create jar, takes long tho)
FROM maven:3.9.6-amazoncorretto-21-al2023 AS build
WORKDIR /home/app
COPY . /home/app
RUN mvn -f /home/app/pom.xml clean package

FROM openjdk:21-jdk-slim
EXPOSE 5000
COPY --from=build /home/app/target/*.jar app.jar
ENTRYPOINT [ "sh", "-c", "java -jar /app.jar" ]
